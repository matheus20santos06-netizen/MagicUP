    <div id="name-modal" class="fixed inset-0 bg-black/80 backdrop-blur-sm hidden items-center justify-center p-4 z-50">
        <div class="bg-slate-900 border border-slate-800 rounded-2xl p-6 w-full max-w-sm shadow-2xl flex flex-col gap-4">
            <h3 class="text-lg font-bold text-white">Nomear Objeto Identificado</h3>
            <p class="text-xs text-slate-400" id="modal-box-info">Dimensões: X=0, Y=0</p>
            <input type="text" id="object-name-input" placeholder="Ex: Chaves, Telemóvel, Garrafa..." class="bg-slate-950 border border-slate-700 rounded-lg px-4 py-2.5 text-sm text-white focus:outline-none focus:border-emerald-500">
            <div class="flex gap-2 justify-end mt-2">
                <button onclick="closeModal()" class="px-4 py-2 bg-slate-800 hover:bg-slate-700 text-slate-300 rounded-lg text-sm transition">Cancelar</button>
                <button onclick="saveObjectName()" class="px-4 py-2 bg-emerald-600 hover:bg-emerald-500 text-white font-bold rounded-lg text-sm transition">Guardar</button>

    <script>
        let currentMode = null;
        let peer = null;
        let conn = null;
        let targetColor = 'red';
        
        // Dados partilhados / detetados
        let lastDetectedBoxes = []; // Guardar caixas ativas para clique
        let activeSelectedBox = null;
        let customLabels = {}; // { "x_y_w_h": "Nome" }

        function initMode(mode) {
            currentMode = mode;
            document.getElementById('mode-selector').classList.add('hidden');
            
            if (mode === 'tablet') {
                document.getElementById('tablet-view').classList.remove('hidden');
                startTabletCamera();
            } else {
                document.getElementById('phone-view').classList.remove('hidden');
                initPhonePeer();
            }
        }

        // --- MODO TABLET ---
        function startTabletCamera() {
            peer = new Peer();
            peer.on('open', id => {
                document.getElementById('my-id').innerText = id;
            });
            peer.on('connection', c => {
                conn = c;
                document.getElementById('connection-status').innerText = 'Ligado ao Telemóvel ✓';
                document.getElementById('connection-status').className = 'px-3 py-1 bg-emerald-500/10 border border-emerald-500/30 text-emerald-400 rounded-full text-xs font-semibold';
                
                conn.on('data', data => {
                    if (data.type === 'changeTarget') {
                        targetColor = data.color;
                    } else if (data.type === 'saveLabel') {
                        customLabels[data.key] = data.name;
                    }
                });
            });

            const video = document.getElementById('video-element');
            const canvas = document.getElementById('canvas-tablet');
            const ctx = canvas.getContext('2d', { willReadFrequently: true });

            navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment', width: { ideal: 640 }, height: { ideal: 480 } } })
                .then(stream => {
                    video.srcObject = stream;
                    video.play();
                    document.getElementById('tablet-overlay-msg').style.display = 'none';
                    requestAnimationFrame(() => processTabletLoop(video, canvas, ctx));
                })
                .catch(err => {
                    document.getElementById('tablet-overlay-msg').innerText = 'Erro ao aceder à câmara: ' + err.message;
                });
        }

        function processTabletLoop(video, canvas, ctx) {
            if (video.readyState === video.HAVE_ENOUGH_DATA) {
                if (canvas.width !== video.videoWidth || canvas.height !== video.videoHeight) {
                    canvas.width = video.videoWidth;
                    canvas.height = video.videoHeight;
                }

                ctx.drawImage(video, 0, 0);
                let frame = ctx.getImageData(0, 0, canvas.width, canvas.height);
                let d = frame.data;
                let minX = canvas.width, maxX = 0, minY = canvas.height, maxY = 0, found = false;

                // Deteção simples por cor dominante
                for (let i = 0; i < d.length; i += 16) { // Amostragem otimizada
                    let r = d[i], g = d[i+1], b = d[i+2];
                    let match = false;
                    if (targetColor === 'red' && r > 140 && r > g + 40 && r > b + 40) match = true;
                    if (targetColor === 'green' && g > 130 && g > r + 30 && g > b + 30) match = true;
                    if (targetColor === 'blue' && b > 130 && b > r + 30 && b > g + 30) match = true;

                    if (match) {
                        let idx = i / 4;
                        let x = idx % canvas.width;
                        let y = Math.floor(idx / canvas.width);
                        minX = Math.min(minX, x); maxX = Math.max(maxX, x);
                        minY = Math.min(minY, y); maxY = Math.max(maxY, y);
                        found = true;
                    }
                }

                let currentBoxes = [];
                if (found && (maxX - minX > 20) && (maxY - minY > 20)) {
                    let boxWidth = maxX - minX;
                    let boxHeight = maxY - minY;
                    // Criar chave aproximada para etiqueta personalizada
                    let boxKey = `${Math.round(minX/50)}_${Math.round(minY/50)}_${Math.round(boxWidth/50)}_${Math.round(boxHeight/50)}`;
                    let label = customLabels[boxKey] || ('Objeto ' + targetColor.toUpperCase());

                    currentBoxes.push({ x: minX, y: minY, w: boxWidth, h: boxHeight, key: boxKey, label: label });

                    // Desenhar retângulo no canvas do tablet
                    ctx.strokeStyle = targetColor === 'red' ? '#ef4444' : targetColor === 'green' ? '#10b981' : '#3b82f6';
                    ctx.lineWidth = 4;
                    ctx.strokeRect(minX, minY, boxWidth, boxHeight);
                    
                    ctx.fillStyle = ctx.strokeStyle;
                    ctx.font = '16px sans-serif';
                    ctx.fillText(label, minX, Math.max(20, minY - 8));
                }

                // Enviar frame atualizado e dados para o telemóvel se conectado
                if (conn && conn.open) {
                    conn.send({
                        image: canvas.toDataURL('image/jpeg', 0.6),
                        boxes: currentBoxes
                    });
                }
            }
            requestAnimationFrame(() => processTabletLoop(video, canvas, ctx));
        }


        // --- MODO TELEMÓVEL ---
        function initPhonePeer() {
            peer = new Peer();
            peer.on('open', id => {
                // Pronto para ligar
            });
        }

        function connectToTablet() {
            let remoteId = document.getElementById('remote-id-input').value.trim();
            if (!remoteId) {
                alert('Insira o ID do Tablet.');
                return;
            }

            conn = peer.connect(remoteId);
            conn.on('open', () => {
                document.getElementById('controller-panel').classList.remove('hidden');
                document.getElementById('phone-overlay-msg').style.display = 'none';
            });

            conn.on('data', data => {
                let canvas = document.getElementById('canvas-phone');
                let ctx = canvas.getContext('2d');
                let img = new Image();
                img.onload = () => {
                    if (canvas.width !== img.width || canvas.height !== img.height) {
                        canvas.width = img.width;
                        canvas.height = img.height;
                    }
                    ctx.clearRect(0, 0, canvas.width, canvas.height);
                    ctx.drawImage(img, 0, 0);

                    lastDetectedBoxes = data.boxes || [];

                    // Redesenhar caixas no telemóvel para feedback visual
                    lastDetectedBoxes.forEach(box => {
                        ctx.strokeStyle = '#10b981';
                        ctx.lineWidth = 4;
                        ctx.strokeRect(box.x, box.y, box.w, box.h);

                        ctx.fillStyle = '#10b981';
                        ctx.font = 'bold 16px sans-serif';
                        ctx.fillRect(box.x, Math.max(0, box.y - 24), ctx.measureText(box.label).width + 16, 24);
                        ctx.fillStyle = '#000';
                        ctx.fillText(box.label, box.x + 8, Math.max(16, box.y - 6));
                    });
                };
                img.src = data.image;
            });

            conn.on('error', err => {
                alert('Erro na ligação: ' + err);
            });
        }

        function changeTarget(color) {
            targetColor = color;
            if (conn && conn.open) {
                conn.send({ type: 'changeTarget', color: color });
            }
        }

        // Clicar no canvas do telemóvel para nomear caixa
        function handleCanvasClick(event) {
            let canvas = document.getElementById('canvas-phone');
            let rect = canvas.getBoundingClientRect();
            let scaleX = canvas.width / rect.width;
            let scaleY = canvas.height / rect.height;
            
            let clickX = (event.clientX - rect.left) * scaleX;
            let clickY = (event.clientY - rect.top) * scaleY;

            // Verificar se o clique está dentro de alguma caixa detetada
            let clickedBox = lastDetectedBoxes.find(b => 
                clickX >= b.x && clickX <= (b.x + b.w) &&
                clickY >= b.y && clickY <= (b.y + b.h)
            );

            if (clickedBox) {
                activeSelectedBox = clickedBox;
                document.getElementById('modal-box-info').innerText = `Objeto detetado em X:${Math.round(clickedBox.x)} Y:${Math.round(clickedBox.y)} (Medida: ${Math.round(clickedBox.w)}x${Math.round(clickedBox.h)})`;
                document.getElementById('object-name-input').value = clickedBox.label.startsWith('Objeto') ? '' : clickedBox.label;
                document.getElementById('name-modal').classList.remove('hidden');
                document.getElementById('name-modal').classList.add('flex');
            }
        }

        function closeModal() {
            document.getElementById('name-modal').classList.add('hidden');
            document.getElementById('name-modal').classList.remove('flex');
            activeSelectedBox = null;
        }

        function saveObjectName() {
            let nameInput = document.getElementById('object-name-input').value.trim();
            if (!nameInput) return;

            if (activeSelectedBox && conn && conn.open) {
                conn.send({
                    type: 'saveLabel',
                    key: activeSelectedBox.key,
                    name: nameInput
                });
            }
            closeModal();
        }
    </script>
